\section{Análise de Grafos com o fork-cli}

\begin{frame}[plain]
  \centering
  \Huge \insertsection
\end{frame}

\begin{frame}{Justificativa do Projeto}
  \begin{itemize}
    \item O aumento recente das tarifas comerciais, especialmente por parte dos Estados Unidos, tem impactado significativamente o comércio internacional.
    \item Essa situação levanta preocupações sobre o efeito das taxações nas relações entre países e na competitividade global.
    \item O projeto busca analisar e compreender como essas tarifas influenciam os fluxos comerciais, auxiliando na tomada de decisões estratégicas.
  \end{itemize}
\end{frame}

\begin{frame}{Fontes dos Dados}
  \begin{itemize}
    \item Os dados de \textbf{Importações} e \textbf{Exportações} utilizados pelo \textbf{fork-cli} são provenientes da \textbf{OCDE}, garantindo confiabilidade e abrangência nas informações comerciais.
    \item As informações sobre \textbf{Tarifas} (baixas e altas) são obtidas do \textbf{WITS}, fonte reconhecida internacionalmente para dados tarifários.
    \item A escolha dessas fontes assegura que as análises realizadas pelo software sejam baseadas em dados oficiais e atualizados.
  \end{itemize}
\end{frame}

\begin{frame}{O que é o fork-cli?}
  \begin{itemize}
    \item O \textbf{fork-cli} é um software de gestão e análise de grafos voltado para o estudo de relações comerciais entre países.
    \item Permite importar, visualizar e analisar diferentes tipos de grafos econômicos.
  \end{itemize}
\end{frame}

\begin{frame}{Tipos de Grafos Analisados}
  O \textbf{fork-cli} irá analisar os seguintes grafos:
  \begin{itemize}
    \item \textbf{Importações}: Relações de importação entre países.
    \item \textbf{Exportações}: Relações de exportação entre países.
    \item \textbf{Superavit}: Países com saldo comercial positivo.
    \item \textbf{Deficit}: Países com saldo comercial negativo.
    \item \textbf{Relações com Baixas Tarifas}: Conexões comerciais com tarifas reduzidas.
    \item \textbf{Relações com Altas Tarifas}: Conexões comerciais com tarifas elevadas.
  \end{itemize}
\end{frame}

\begin{frame}{Como o fork-cli Analisa os Grafos}
  \begin{itemize}
    \item O software constrói grafos a partir dos dados comerciais.
    \item Permite identificar padrões, como países com superavit ou deficit.
    \item Analisa o impacto das tarifas nas relações comerciais.
    \item Gera visualizações e relatórios para apoiar decisões estratégicas.
  \end{itemize}
\end{frame}

\begin{frame}{Exemplo de Visualização}
  \begin{itemize}
    \item Visualização de países conectados por relações de exportação e importação.
    \item Destaque para conexões com tarifas altas ou baixas.
    \item Identificação de clusters de países com superavit ou deficit.
  \end{itemize}
\end{frame}
