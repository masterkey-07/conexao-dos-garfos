\begin{frame}{O Problema}
\begin{itemize}
  \item O comércio internacional é uma rede complexa.
  \item Hoje existem 195 países no planeta.
  \item Como encontrar uma forma eficiente de se estudar essas relações?
\end{itemize}
\end{frame}

\begin{frame}{Complexidade}
\begin{itemize}
  \item Países mantém um grande número de relações entre sí.
  \item Relações comerciais possuem um caráter instável. 
  \item Existem diversas formas que dois países podem se relacionar que precisam ser analisadas.
\end{itemize}
\end{frame}

\begin{frame}{Por que resolver com Grafos?}
\begin{itemize}
  \item Países serão vértices.
  \item As relações comerciais serão as arestas.
  \item Permite rápida análise dessas relações e de outros dados relevantes.
\end{itemize}
\end{frame}

\begin{frame}{O Modelo}
\begin{itemize}
  \item Serão utilizados 5 grafos diferentes para compor nosso modelo.
  \item O primeiro possui como aresta a relação de balança comercial entre os países. 
  \item Os outros três possuem como arestas a relação tarifária entre os países
  \item Por fim, serão criados novos grafos através da operação de intersecção dos anteriores.
\end{itemize}
\end{frame}
